% fig8_capture.tex -- where an onset zone of radius s can sit in X and still
% be seen whole, be missed, or be seen twice.  Triangle rule; zones measured
% between triangle centroids.  Data: results/soz_capture.json
% (run_soz_capture.py) via make_ecog_paper_data.py.
\documentclass[tikz,border=4pt]{standalone}
\input{ecogstyle}
\input{../numbers_ecog.tex}
\begin{document}
\begin{tikzpicture}
\begin{groupplot}[ecog,
  group style={group size=3 by 1, horizontal sep=1.45cm},
  width=6.1cm, height=5.6cm,
  xmin=0, xmax=15, xtick={0,5,10,15},
  xlabel={onset-zone radius $s$ (mm)},
  legend style={at={(0.5,-0.27)}, anchor=north, font=\scriptsize},
  every axis plot/.append style={line width=1.5pt},
]
% ---------------------------------------------------------------- (a)
\nextgroupplot[ymin=0, ymax=106, ytick={0,20,...,100},
  ylabel={onset-zone positions (\%)},
  title={seen whole ($r=8$\,mm)}]
  \addplot[cOrange] table[x=s, y expr=100*\thisrow{whole}] {data/capture_GridEight.dat};
  \addlegendentry{$8\times8$ grid, 64}
  \addplot[cAqua, mk, mark repeat=4] table[x=s, y expr=100*\thisrow{whole}] {data/capture_CrownEight.dat};
  \addlegendentry{crown sites, 64}
  \addplot[cBlue, dash pattern=on 4pt off 2pt] table[x=s, y expr=100*\thisrow{whole}] {data/capture_FreeSixtyFour.dat};
  \addlegendentry{placed, 64}
  \addplot[cBlue] table[x=s, y expr=100*\thisrow{whole}] {data/capture_OneEight.dat};
  \addlegendentry{placed, \capNOneEight{} ($X\subseteq R_1$)}
% ---------------------------------------------------------------- (b)
\nextgroupplot[ymin=0, ymax=60, ytick={0,10,...,60},
  title={missed entirely}]
  \addplot[cOrange] table[x=s, y expr=100*\thisrow{missed}] {data/capture_GridEight.dat};
  \addlegendentry{$8\times8$ grid, $r=8$\,mm}
  \addplot[cOrange, dash pattern=on 4pt off 2pt] table[x=s, y expr=100*\thisrow{missed}] {data/capture_GridTen.dat};
  \addlegendentry{$8\times8$ grid, $r=10$\,mm}
  \addplot[cAqua, mk, mark repeat=4] table[x=s, y expr=100*\thisrow{missed}] {data/capture_CrownEight.dat};
  \addlegendentry{crown sites, 64, $r=8$\,mm}
  \addplot[cBlue, dash pattern=on 4pt off 2pt] table[x=s, y expr=100*\thisrow{missed}] {data/capture_FreeSixtyFour.dat};
  \addlegendentry{placed, 64, $r=8$\,mm}
  \node[font=\scriptsize, text=cInk2, align=right, anchor=north east]
    at (rel axis cs:0.97,0.97) {designs with $X\subseteq R_1$: $0$};
% ---------------------------------------------------------------- (c)
\nextgroupplot[ymin=0, ymax=106, ytick={0,20,...,100},
  title={seen twice}]
  \addplot[cBlue] table[x=s, y expr=100*\thisrow{twice}] {data/capture_OneEight.dat};
  \addlegendentry{placed, \capNOneEight, $r=8$\,mm ($k=1$)}
  \addplot[cBlue, dash pattern=on 4pt off 2pt] table[x=s, y expr=100*\thisrow{twice}] {data/capture_OneTen.dat};
  \addlegendentry{placed, \capNOneTen, $r=10$\,mm ($k=1$)}
  \addplot[cViolet!40, line width=4.5pt] table[x=s, y expr=100*\thisrow{twice}] {data/capture_TwoTen.dat};
  \addlegendentry{placed, \capNTwoTen, $r=10$\,mm ($k=2$)}
  \addplot[cViolet, mksq, mark repeat=5] table[x=s, y expr=100*\thisrow{twice}] {data/capture_TwoEight.dat};
  \addlegendentry{placed, \capNTwoEight, $r=8$\,mm ($k=2$)}
  \node[font=\scriptsize, text=cInk2, anchor=north east, align=right]
    at (rel axis cs:0.97,0.90) {both $k=2$ designs:\\$100\%$ at every $s$};
\end{groupplot}
\node[panellabel] at ($(group c1r1.north west)+(-1.15,0.62)$) {(a)};
\node[panellabel] at ($(group c2r1.north west)+(-0.95,0.62)$) {(b)};
\node[panellabel] at ($(group c3r1.north west)+(-0.95,0.62)$) {(c)};
\end{tikzpicture}
\end{document}
