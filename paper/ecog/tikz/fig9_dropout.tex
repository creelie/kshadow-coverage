\newcommand{\dplot}[4]{% #1 style, #2 design, #3 s, #4 legend
  \addplot[#1, error bars/.cd, y dir=minus, y explicit,
    error bar style={line width=0.7pt, solid}, error mark options={draw=none}]
    table[x=q, y expr=100*\thisrow{#2_s#3_mean},
      y error minus expr=100*(\thisrow{#2_s#3_mean}-\thisrow{#2_s#3_p05})]
    {data/dropout.dat};
  \addlegendentry{#4}}
